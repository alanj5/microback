cmake_minimum_required(VERSION 3.24)
project(microback VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
option(MICROBACK_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(MICROBACK_WARNINGS_AS_ERRORS "Treat project warnings as errors" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

add_library(microback_options INTERFACE)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(microback_options INTERFACE -Wall -Wextra -Wconversion -Wsign-conversion -Wshadow)
  if(MICROBACK_WARNINGS_AS_ERRORS)
    target_compile_options(microback_options INTERFACE -Werror)
  endif()
  if(MICROBACK_SANITIZERS)
    target_compile_options(microback_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(microback_options INTERFACE -fsanitize=address,undefined)
  endif()
elseif(MICROBACK_SANITIZERS)
  message(FATAL_ERROR "Sanitizer configuration requires GCC or Clang")
endif()

add_library(microback_core src/order_book.cpp src/event.cpp src/replay_engine.cpp src/execution.cpp src/portfolio.cpp src/risk.cpp src/strategy.cpp)
target_include_directories(microback_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(microback_core PUBLIC microback_options)

add_executable(microback_cli apps/replay.cpp)
target_link_libraries(microback_cli PRIVATE microback_core)

if(BUILD_TESTING)
  include(FetchContent)
  FetchContent_Declare(Catch2
    GIT_REPOSITORY https://github.com/catchorg/Catch2.git
    GIT_TAG v3.7.1
    GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(Catch2)
  add_executable(microback_tests tests/cpp/test_types.cpp tests/cpp/test_order_book.cpp tests/cpp/test_replay.cpp tests/cpp/test_execution.cpp tests/cpp/test_portfolio.cpp tests/cpp/test_strategy.cpp)
  target_compile_definitions(microback_tests PRIVATE MICROBACK_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  target_link_libraries(microback_tests PRIVATE microback_core Catch2::Catch2WithMain)
  list(APPEND CMAKE_MODULE_PATH ${catch2_SOURCE_DIR}/extras)
  include(Catch)
  catch_discover_tests(microback_tests)
  add_test(NAME canonical_cli_fixture COMMAND microback_cli ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/replay.csv)
  set_tests_properties(canonical_cli_fixture PROPERTIES PASS_REGULAR_EXPRESSION "events=9 snapshots=1 sequence_gaps=0")
endif()
